// Regenerates app/(store)/components/scenes/pacific-data.ts from Natural Earth (world-atlas). Run in a scratch dir with: npm i world-atlas topojson-client d3-geo, then node gen.mjs and copy pacific-data.ts back.
import { readFileSync, writeFileSync } from "node:fs";
import { feature } from "topojson-client";
import { geoEquirectangular, geoPath, geoGraticule } from "d3-geo";
const load = (f) => JSON.parse(readFileSync(`node_modules/world-atlas/${f}`, "utf8"));
const c10 = load("countries-10m.json"), l10 = load("land-10m.json"), l50 = load("land-50m.json");
const nations = feature(c10, c10.objects.countries).features;
const MIC = { Palau: "palau", Micronesia: "fsm", "Marshall Is.": "marshall", Kiribati: "kiribati", Nauru: "nauru", Guam: "guam", "N. Mariana Is.": "cnmi" };
const micFeatures = nations.filter((f) => MIC[f.properties.name]);

function view(lon0, lon1, lat0, lat1, width, landTopo, landObj) {
  // rotate so the dateline is the middle: x increases eastward across it
  const cx = (lon0 + lon1) / 2;
  const projection = geoEquirectangular().rotate([-cx, 0]).scale(1).translate([0, 0]);
  const w = lon1 - lon0, h = lat1 - lat0;
  const k = width / (w * Math.PI / 180);
  projection.scale(k).translate([width / 2, (lat1 * Math.PI / 180) * k]);
  projection.clipExtent([[0, 0], [width, h * Math.PI / 180 * k]]);
  const path = geoPath(projection).digits?.(1) ?? geoPath(projection);
  const round = (d) => d.replace(/(\d+\.\d{2,})/g, (m) => (+m).toFixed(1));
  const land = round(path(feature(landTopo, landTopo.objects[landObj])) ?? "");
  const nationPaths = {};
  for (const f of micFeatures) nationPaths[MIC[f.properties.name]] = round(path(f) ?? "");
  const pt = (lon, lat) => { const p = projection([lon, lat]); return [+p[0].toFixed(1), +p[1].toFixed(1)]; };
  const grat = round(path(geoGraticule().step([10, 10]).extent([[lon0 - 90 > -180 ? lon0 - 90 : -180, lat0], [lon1, lat1]])()) ?? "");
  return { width, height: +(h * Math.PI / 180 * k).toFixed(1), land, nations: nationPaths, pt, grat };
}
const l110 = load("land-110m.json"); const wide = view(118, 246, -14, 54, 1000, l110, "land");
const close = view(128, 180, -6, 22, 1000, l50, "land");
const P = (v, name, lon, lat) => [name, ...v.pt(lon, lat)];
const out = {
  wide: { grat: wide.grat, zoom: [...wide.pt(128, 22), ...wide.pt(180, -6)], width: wide.width, height: wide.height, land: wide.land, nations: wide.nations, lacey: wide.pt(-122.8, 47.0) },
  close: { grat: close.grat, width: close.width, height: close.height, land: close.land, nations: close.nations,
    places: [
      ["Palau", 134.5, 7.5, "w"], ["Yap", 138.1, 9.5, "n"], ["Guam", 144.8, 13.45, "s"], ["Saipan", 145.7, 15.2, "n"],
      ["Chuuk", 151.8, 7.4, "s"], ["Pohnpei", 158.2, 6.9, "n"], ["Kosrae", 163.0, 5.3, "s"], ["Nauru", 166.9, -0.52, "e"],
      ["Majuro", 171.4, 7.1, "n"], ["Tarawa", 173.0, 1.4, "s"],
    ].map(([n, lon, lat, side]) => ({ name: n, x: close.pt(lon, lat)[0], y: close.pt(lon, lat)[1], side })),
    labels: [
      ["PALAU", 134.5, 5.2, "end"], ["FEDERATED STATES OF MICRONESIA", 150, 3.2, "middle"], ["MARIANA ISLANDS", 148.6, 18.2, "start"],
      ["MARSHALL ISLANDS", 168.2, 10.6, "middle"], ["KIRIBATI", 174, -2.4, "start"], ["NAURU", 167, -2.6, "end"],
    ].map(([t, lon, lat, anchor]) => ({ text: t, x: close.pt(lon, lat)[0], y: close.pt(lon, lat)[1], anchor })) },
};
const ts = `/* Generated by scripts (Natural Earth via world-atlas, public domain). Do not edit by hand. */\nexport const PACIFIC = ${JSON.stringify(out)} as const;\n`;
writeFileSync("pacific-data.ts", ts);
console.log("bytes", ts.length, "wide land", wide.land.length, "close land", close.land.length, Object.keys(out.close.nations));
